Hints06B

Task 6 · Hints and solution

Create a RecipeListViewModel

  • public bool IsFiltered => Cuisine is not null; stores nothing.
  • View("Index", vm) names the view: ByCuisine has none of its own.
  • Store the cuisine as the data spells it: matches.FirstOrDefault()?.Cuisine ?? cuisine.
  • using RecipeBox.Models.ViewModels; in the controller.

Controllers/RecipesController.cs

[HttpGet("")]
public IActionResult Index(string? cuisine) => View(BuildList(cuisine));

[HttpGet("cuisine/{name}")]
public IActionResult ByCuisine(string name) => View("Index", BuildList(name));

private RecipeListViewModel BuildList(string? cuisine)
{
    var matches = string.IsNullOrWhiteSpace(cuisine)
        ? _recipes
        : _recipes.Where(r => r.Cuisine.ToLower() == cuisine.ToLower()).ToList();

    return new RecipeListViewModel
    {
        Recipes    = matches.OrderBy(r => r.Title).ToList(),
        TotalCount = _recipes.Count,
        Cuisine    = string.IsNullOrWhiteSpace(cuisine)
                         ? null : matches.FirstOrDefault()?.Cuisine ?? cuisine
    };
}
