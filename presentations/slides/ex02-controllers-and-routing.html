Task 1 · Hints and solution

Add a RecipesController with an Index action

  • using RecipeBox.Models; at the top, or Recipe is an unknown type.
  • The URL is /Recipes: the framework strips the word Controller. Index is the default action.
  • static keeps one list for the whole app, not a new one per request.
using Microsoft.AspNetCore.Mvc;
using RecipeBox.Models;

namespace RecipeBox.Controllers;

public class RecipesController : Controller
{
    // In-memory for now. Set 4 moves it into a database.
    private static readonly List<Recipe> _recipes = new()
    {
        new Recipe { Id = 1, Title = "Pancakes", Cuisine = "American",
                     PrepMinutes = 10, CookMinutes = 15, Servings = 4,
                     Ingredients = new() { "Flour", "Milk", "Eggs", "Sugar" } },
        // ... recipes 2–5, copied from RecipeBoxBasics
    };

    // GET /Recipes
    public IActionResult Index()
    {
        return Content($"RecipeBox has {_recipes.Count} recipes.");
    }
}

Task 2 · Hints and solution

Return the catalog as plain text and as JSON

  • Select turns each recipe into a string; string.Join("\n", lines) glues them.
  • Do not name the action Json: that is the method we are calling.
  • Lower-case names are the JSON convention. The framework converts them.
// GET /Recipes
public IActionResult Index()
{
    var lines = _recipes.Select(r => $"{r.Id}: {r.Title}");
    return Content($"{_recipes.Count} recipe(s)\n" + string.Join("\n", lines));
}

// GET /Recipes/All
public IActionResult All() => Json(_recipes);

Task 4 · Hints and solution

Return 404 for an unknown recipe

  • This is Set 1's Banitsa question, now with a status code attached.
  • is null reads better than == null and means the same here.
  • The message makes Task 8's difference visible.
// GET /Recipes/Details/2
public IActionResult Details(int id)
{
    var recipe = _recipes.FirstOrDefault(r => r.Id == id);
    if (recipe is null)
        return NotFound($"No recipe with id {id}.");

    return Content($"{recipe.Title} ({recipe.Cuisine})\n" +
                   $"{recipe.TotalMinutes()} min, serves {recipe.Servings}\n" +
                   $"Ingredients: {string.Join(", ", recipe.Ingredients)}");
}

Hints05A

Task 5 · Hints and solution

Filter by cuisine with a query-string parameter

  • ?cuisine=Italian fills string? cuisine because the names match.
  • string.IsNullOrWhiteSpace(cuisine) is true for missing, empty or blank.
  • Both sides of ? : must be a List<Recipe>: end the Where with .ToList().
// GET /Recipes            GET /Recipes?cuisine=Italian
public IActionResult Index(string? cuisine)
{
    var matches = string.IsNullOrWhiteSpace(cuisine)
        ? _recipes
        : _recipes.Where(r => r.Cuisine == cuisine).ToList();

    var lines = matches.Select(r => $"{r.Id}: {r.Title}");
    return Content($"{matches.Count} recipe(s)\n" + string.Join("\n", lines));
}

Hints07B

Task 7 · Hints and solution

Give each action a clean URL with attribute routing

  • [Route] on the class is a prefix; [HttpGet("…")] completes it. "" means the prefix itself.
  • A leading / escapes the prefix: [HttpGet("/about")].
  • Once a controller has attribute routes, the default route no longer reaches it.
  • The query string is not part of the template: Index needs no change.
[Route("recipes")]
public class RecipesController : Controller
{
    // ... _recipes unchanged ...

    [HttpGet("")]            // GET /recipes   and   /recipes?cuisine=Italian
    public IActionResult Index(string? cuisine) { ... }

    [HttpGet("json")]        // GET /recipes/json
    public IActionResult All() => Json(_recipes);

    [HttpGet("{id}")]        // GET /recipes/2
    public IActionResult Details(int id) { ... }

    [HttpGet("/about")]      // GET /about
    public IActionResult About() => ...;
}
/Recipes/Details/2  →  404   # attribute-routed now: the default
                              # route no longer reaches this controller

Hints09B

Task 9 · Hints and solution

Add a by-cuisine route

  • r.Cuisine.ToLower() == name.ToLower() compares case-blind.
  • Why 404 here, but 200 in Task 5? A path names a thing; a query string is a search.
  • {name} has no ?, so it is required.
// GET /recipes/cuisine/italian
[HttpGet("cuisine/{name}")]
public IActionResult ByCuisine(string name)
{
    var titles = _recipes
        .Where(r => r.Cuisine.ToLower() == name.ToLower())
        .Select(r => r.Title)
        .ToList();

    if (titles.Count == 0)
        return NotFound($"No {name} recipes.");

    return Content($"{name}: {string.Join(", ", titles)}");
}

Hints10B

Task 10 · Hints and solution

Generate links with Url.Action

  • new { id = r.Id }: the property name must match the route parameter, id.
  • Url.Action reads our routes, so it knows the template is recipes/{id:int}.
  • In Set 3 the same idea appears in views as asp-action and asp-route-id.
[HttpGet("")]
public IActionResult Index(string? cuisine)
{
    var matches = string.IsNullOrWhiteSpace(cuisine)
        ? _recipes
        : _recipes.Where(r => r.Cuisine == cuisine).ToList();

    var lines = matches.Select(r =>
        $"{r.Id}: {r.Title} -> {Url.Action("Details", new { id = r.Id })}");
    return Content($"{matches.Count} recipe(s)\n" + string.Join("\n", lines));
}
